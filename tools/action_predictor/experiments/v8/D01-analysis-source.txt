from pathlib import Path

import sys,json
import numpy as np
root=Path(__file__).resolve().parent.parent;sys.path.insert(0,str(root))
from tools.action_predictor.event_semantics import PLAN,sources

report=read(REPORTS/'D01.json');assert report['plan_sha256']==sha(PLAN)
assert read(PLAN)['source_hashes']==report['source_hashes']==sources()
seen=np.zeros(433071,dtype=bool)
for item in report['bindings']:
    start=item['offset'];stop=start+item['rows'];assert 0<=start<stop<=len(seen) and not seen[start:stop].any()
    seen[start:stop]=True
assert seen.all()
totals=Counter()
for data in report['statistics'].values():totals.update(data)
with np.load(BASE/'combined-train.npz') as z:labels=z['y']
from tools.action_predictor import reliable_train as rt
for i,name in enumerate(rt.LABELS):assert totals['target/'+name]==int((labels==i).sum())
previous=read(REPORTS/'A01-state-and-generalization-diagnostic.json')['training_contingency']['per_action']
for name in ('end','cancel','priority','recall','ward'):
    for key,before in [('equal','known_equal'),('different','known_different'),('unknown','unknown')]:
        assert totals['target_priority_relation/'+name+'/'+key]==previous[name][before]

static={}
for path,expected in report['static_source_hashes'].items():
    assert sha(path)==expected
    if Path(path).name in ('PanelChrome-DRtW3oWL.js','index-B7p3Cpbd.js'):
        text=Path(path).read_text(encoding='utf-8')
        needles=['Yielded=0','FuelExhausted=1','Error=2'] if Path(path).name.startswith('PanelChrome') else ['finishTurn(e=F.Yielded)','endTurn(e){for(this.setPriority(e)','cancelChannel(){this.check(this.wasmExports.Game_Action_CancelChannel']
        static[path]=dict(sha256=expected,evidence=[])
        for needle in needles:
            offset=text.find(needle);assert offset>=0
            static[path]['evidence'].append(dict(needle=needle,offset=offset,snippet=text[max(0,offset-50):offset+240]))
probe_table=[]
for probe in report['probes']:
    assert sha(probe['query_path'])==probe['query_sha256'] and sha(probe['output_path'])==probe['output_sha256']
    for row in probe['data']['rows']:
        assert row['pre_header']['phase']==10 and row['pre_header']['currentChampion']==row['actor']
        after=row['post_header'];step=after['step']
        probe_table.append(dict(engine=probe['engine'],kind=step['kind'],actor=row['actor'],pre=row['pre'],event=row['event'],before=row['before'],after=row['after'],post_phase=after['phase'],reason=step.get('endTurnReason'),automatic_headers_observed=sum(h['step']['substep']>0 for h in row['following'])))
result=dict(report_sha256=sha(REPORTS/'D01.json'),plan_sha256=sha(PLAN),source_sha256=sha(__file__),static_evidence=static,totals=dict(sorted(totals.items())),probes=probe_table,
    row_alignment_verified=True,all433071_labels_verified=True,A01_priority_relation_counts_reproduced=True,
    conclusions=['The pinned codec names end reason0 Yielded. The extractor historical completeness name explicit_end does not establish whether the bot explicitly called end_turn; origin remains unassessed. Frozen membership/labels are unchanged.',
    'Root substep0 phase10 champion events are separate from automatic substeps. Header phase AFTER the target, current target kind/reason and POST channel changes are diagnostic labels/evidence, forbidden as predictor inputs.',
    'A priority setter and following end are separate recorded inputs. A frequent observed sequence is not a legality constraint. Use any PRE priority/channel cue only as a separately tested soft input.',
    'Current SDK4 and visualiser wrapper show API names and compound operations, but cannot establish historical engine2/3 bot caller behavior. Engine-specific recorded PRE/POST probes are kept as narrower direct evidence.',
    'No predictor, accuracy, strategy or generalization improvement is measured by this diagnostic. The P05 failed gate and all original targets remain binding.'],
    new_features=0,new_optimizer_steps=0,new_prediction_families=0,reserved_behavior_opened=0)
frozen(REPORTS/'D01-analysis.json',result)
(REPORTS/'D01-analysis-source.txt').write_bytes(Path(__file__).read_bytes())
